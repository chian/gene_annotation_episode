class Source:
    def __init__(self, *, current="1.8", current_unit="pA", voltage="−200", voltage_unit="mV",
                 reversal="53", reversal_unit="mV", conductance="7", conductance_unit="pS",
                 current_prefix="", current_suffix="", reversal_spread=None):
        self.reference = "source_" + "a"*24
        self.covariance_ref = "source_" + "b"*24
        self.tokens = dict(current=current, applied_voltage=voltage, reversal_voltage=reversal,
                           reported_conductance=conductance)
        self.units = dict(current=current_unit, applied_voltage=voltage_unit,
                          reversal_voltage=reversal_unit, reported_conductance=conductance_unit)
        self.cells = {role: f"<td>{current_prefix if role == 'current' else ''}{token}"
                     f"{current_suffix if role == 'current' else ''}"
                     f"{' ± ' + reversal_spread if role == 'reversal_voltage' and reversal_spread else ''}</td>"
                     for role, token in self.tokens.items()}
        headers = ''.join(f"<th>{role} ({self.units[role]})</th>" for role in QUANTITIES)
        self.text = ("<article><p>Synthetic membrane experiment. Same applicable ionic condition. "
                     "Current-step magnitude; signed current; single preparation; closed-channel baseline. "
                     "An unrelated biionic reversal was 119 mV. Spread meaning unspecified. "
                     "Known synthetic primitive covariance.</p><table><tr>" + headers + "</tr><tr>"
                     + ''.join(self.cells.values()) + "</tr></table></article>")
        self.content = self.text.encode()
        self.artifact = "artifact_" + hashlib.sha256(self.content).hexdigest()
        self.calls = []
        self.covariance = None

    def witness(self, token, *, occurrence=0):
        start = -1
        for _ in range(occurrence+1):
            start = self.text.index(token, start+1)
        return {"source_ref": self.reference, "representation": "original", "quote": token,
                "start": len(self.text[:start].encode()), "end": len(self.text[:start+len(token)].encode())}

    def number(self, token, *, container=None):
        container = token if container is None else container
        start = container.index(token)
        return {"witness": self.witness(container), "number_start": start, "number_end": start+len(token)}

    def interpreted(self, value, quote=None):
        return {"value": value, "witnesses": [] if quote is None else [self.witness(quote)]}

    def args(self):
        measurements = {}
        cell_start = self.text.index('</tr><tr>') + len('</tr><tr>')
        for role in QUANTITIES:
            header = f"<th>{role} ({self.units[role]})</th>"
            uw = self.witness(header)
            index = header.index(self.units[role])
            uw.update(start=uw["start"]+len(header[:index].encode()),
                      end=uw["start"]+len(header[:index+len(self.units[role])].encode()), quote=self.units[role])
            number = self.number(self.tokens[role], container=self.cells[role])
            number['witness'].update(start=len(self.text[:cell_start].encode()),
                                     end=len(self.text[:cell_start+len(self.cells[role])].encode()))
            cell_start += len(self.cells[role])
            measurements[role] = {"number": number,
                "unit": uw, "context": self.witness("Same applicable ionic condition"), "spread": None,
                "spread_kind": self.interpreted("unknown")}
        return {"source_ref": self.reference, "source_kind": "synthetic_fixture",
                "literals": {name: None for name in LITERAL_FIELDS},
                "arrangement_kind": self.interpreted("single_preparation", "single preparation"),
                "current_kind": self.interpreted("current_step_magnitude", "Current-step magnitude"),
                "arrangement_relation": self.interpreted("same_electrical_arrangement_supported", "Same applicable ionic condition"),
                "measurements": measurements, "covariance_source_ref": None,
                "variance_status": "unknown", "variance_witnesses": []}

    def read(self, reference, representation):
        self.calls.append((reference, representation))
        assert representation == "original"
        if reference == self.covariance_ref:
            raw = self.covariance if isinstance(self.covariance, bytes) else json.dumps(self.covariance).encode()
            return {"artifact_ref": "artifact_" + hashlib.sha256(raw).hexdigest(), "content": raw}
        assert reference == self.reference
        return {"artifact_ref": self.artifact, "content": self.content}


def run(source, args=None):
    extraction = read_channel_inputs(source.args() if args is None else args, read_source=source.read)
    return extraction, populate_channel_components(channel_calculation_inputs(extraction))


def inputs(**changes):
    result = {"values": dict(current=1.8, applied_voltage=-200, reversal_voltage=53, reported_conductance=7),
        "exact_values": {name: None for name in QUANTITIES},
        "measurement_keys": {name: "measurement_" + name for name in QUANTITIES},
        "current_kind": "current_step_magnitude", "arrangement_relation": "same_electrical_arrangement_supported",
        "covariance": None, "variance_status": "unknown", "origin": "synthetic"}
    result.update(changes)
    return result


def assert_error(code, function, *args, **kwargs):
    with pytest.raises(ExperimentalContractError) as caught:
        function(*args, **kwargs)
    assert caught.value.code == code


def test_exact_source_chord_and_separate_reported_value():
    source = Source()
    extraction, sheet = run(source)
    cells = sheet["cells"]
    assert set(cells) == set(COMPONENTS)
    assert cells["chord_conductance"]["value"] == float(Fraction(1800,253))
    assert cells["reported_conductance"]["value"] == 7
    assert cells["absolute_driving_voltage"]["value"] == 253
    assert cells["charge_equivalent_rate"]["value"] == float(Fraction("1.8e-12")/Fraction("1.602176634e-19"))
    assert cells["charge_equivalent_rate"]["unit"] == "elementary_charge_equivalents_per_s"
    assert cells["chord_conductance_variance"]["unavailable_reason"] == "primitive_covariance_not_supplied"
    assert cells["chord_conductance"]["operand_roles"] == list(PRIMITIVE_ORDER)
    assert len(cells["chord_conductance"]["measurement_keys"]) == 3
    assert extraction["values"]["applied_voltage"] == -200
    assert source.calls == [(source.reference, "original")]
    assert cells["current"]["origin"] == "synthetic"
    assert cells["chord_conductance"]["origin"] == "derived"


@pytest.mark.parametrize("role,token,unit,expected", [
    ("current", "0.0018", "nA", 1.8), ("current", "1.8e-12", "A", 1.8),
    ("current", "1.8", "pA", 1.8), ("applied_voltage", "−0.2", "V", -200),
    ("reversal_voltage", "0.053", "V", 53), ("reported_conductance", "0.007", "nS", 7),
    ("reported_conductance", "7e-12", "S", 7), ("reported_conductance", "7", "pS", 7)])
def test_exact_supported_unit_conversions(role, token, unit, expected):
    keywords = {"current": ("current", "current_unit"), "applied_voltage": ("voltage", "voltage_unit"),
                "reversal_voltage": ("reversal", "reversal_unit"),
                "reported_conductance": ("conductance", "conductance_unit")}
    k1,k2 = keywords[role]
    extraction, _ = run(Source(**{k1:token,k2:unit}))
    assert extraction["values"][role] == expected
    exact = extraction["exact_values"][role]
    assert Fraction(int(exact["numerator"]),int(exact["denominator"])) == Fraction(str(expected))

@pytest.mark.parametrize("sign_i,sign_d", [(1,1),(1,-1),(-1,1),(-1,-1)])
def test_full_covariance_gradient_has_correct_absolute_value_signs(sign_i,sign_d):
    data = inputs(current_kind="signed_current", covariance=[[.04,.01,.02],[.01,1,.5],[.02,.5,4]],
                  variance_status="known")
    data["values"].update(current=2*sign_i, applied_voltage=2+8*sign_d, reversal_voltage=2)
    cells = populate_channel_components(data)["cells"]
    gradient = (Fraction(125*sign_i), Fraction(-125*sign_d,4), Fraction(125*sign_d,4))
    matrix = [[Fraction(str(v)) for v in row] for row in data["covariance"]]
    expected = sum(gradient[i]*matrix[i][j]*gradient[j] for i in range(3) for j in range(3))
    assert cells["chord_conductance"]["value"] == 250
    assert cells["chord_conductance_variance"]["value"] == float(expected)
    assert cells["chord_conductance_standard_uncertainty"]["value"] == pytest.approx(math.sqrt(float(expected)))

def covariance_args(source):
    args = source.args()
    extraction = read_channel_inputs(args,read_source=source.read)
    source.covariance = {"version":"a105_channel_covariance_v1", "primitive_order":list(PRIMITIVE_ORDER),
        "primitive_measurement_keys":[extraction["measurement_keys"][name] for name in PRIMITIVE_ORDER],
        "primitive_units":["pA","mV","mV"], "covariance":[[.04,0,0],[0,1,0],[0,0,4]],"origin":"synthetic"}
    args.update(covariance_source_ref=source.covariance_ref,variance_status="known",
                variance_witnesses=[source.witness("Known synthetic primitive covariance")])
    return args


def test_source_covariance_has_exact_primitive_owner_order_and_no_extra_reads():
    source=Source()
    args=covariance_args(source)
    source.calls.clear()
    extraction,sheet=run(source,args)
    assert extraction["covariance_origin"] == "synthetic"
    assert sheet["cells"]["chord_conductance_variance"]["value"] > 0
    assert source.calls == [(source.reference,"original"),(source.covariance_ref,"original")]
